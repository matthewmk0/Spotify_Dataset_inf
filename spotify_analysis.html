Chi-Square Test of Independence (genre vs danceability)

chi_sp <- filter(clean_sp, genre != "Pop")
dance_cat <- ifelse(chi_sp$danceability > median(chi_sp$danceability),
                    "Danceable", "Not Danceable")

# Observed counts
